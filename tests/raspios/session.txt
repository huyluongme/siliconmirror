# A Raspberry Pi OS session on the raspi3b: the card of
# scripts/raspios-card.sh, the release image as Raspberry Pi Imager
# writes it (a user-data that makes the user pi and turns the serial
# console on), which the firmware boots as a real card. drive.c runs
# it; see its comment for the commands.
#
# The first boot says nothing on the serial console, as enable_uart is
# 0: Raspberry Pi OS grows the root partition and its file system,
# cloud-init makes the user, and raspi-config sets enable_uart=1, then
# the board reboots: most of the ten minutes of the session.
timeout 3600
expect Booting Linux on physical CPU 0x0000000000 [0x410fd034]
expect Machine model: Raspberry Pi 3 Model B Rev 1.2
expect smp: Brought up 1 node, 4 CPUs
expect mmcblk0: mmc0:aaaa SMSD1 8.00 GiB
expect EXT4-fs (mmcblk0p2): mounted filesystem
expect Debian GNU/Linux 13 raspberrypi ttyS0
expect raspberrypi login:
timeout 300
type pi\r
expect Password:
type raspberry\r
expect pi@raspberrypi:~$

# The board, the grown root partition (from sector 1064960 to the end
# of the card of 8 GiB), and what cloud-init and raspi-config did.
type cat /proc/device-tree/model; echo\r
expect Raspberry Pi 3 Model B Rev 1.2
expect pi@raspberrypi:~$
type cat /sys/block/mmcblk0/mmcblk0p2/size\r
expect 15712256
expect pi@raspberrypi:~$
type id\r
expect uid=1000(pi) gid=1000(pi)
expect pi@raspberrypi:~$
type grep --color=never enable_uart /boot/firmware/config.txt\r
expect enable_uart=1
expect pi@raspberrypi:~$

# vcgencmd, through /dev/vcio and the firmware's GET_GENCMD_RESULT.
type vcgencmd version\r
expect Copyright (c) 2012 Broadcom
expect version 13c4219f16dbe58c5b9946086b474baf572512bb (clean) (release) (start)
expect pi@raspberrypi:~$
type vcgencmd measure_temp\r
expect temp=45.1'C
expect pi@raspberrypi:~$
type vcgencmd measure_clock arm\r
expect frequency(48)=
expect pi@raspberrypi:~$

# The MMAL users, through VCHIQ's 'mmal': bcm2835-codec's five devices,
# bcm2835-isp's two instances of four, and no camera (no /dev/video0).
# Then what V4L2 lists of each (VIDIOC_ENUM_FMT, 0xc0405602, by
# python3): the device, the buffer type, the count and the fourccs.
type echo /dev/video*\r
expect /dev/video10 /dev/video11 /dev/video12 /dev/video13 /dev/video14 /dev/video15 /dev/video16 /dev/video18 /dev/video20 /dev/video21 /dev/video22 /dev/video23 /dev/video31
expect pi@raspberrypi:~$
type python3 -c "exec('import fcntl,struct\\nfor d,t in ((10,10),(10,9),(11,10),(11,9),(12,10),(12,9),(13,2),(14,1),(15,1),(16,13),(18,10),(18,9),(31,10),(31,9)):\\n f=open(\\'/dev/video%d\\'%d,\\'rb\\');n=[]\\n try:\\n  while 1:n.append(fcntl.ioctl(f,0xc0405602,struct.pack(\\'II\\',len(n),t)+bytes(56))[44:48].decode().strip())\\n except OSError:print(\\'fmt\\',d,t,len(n),*n)')"\r
expect fmt 10 10 4 MPG4 H264 H263 MJPG
expect fmt 10 9 5 YU12 YV12 NV12 NV21 RGBP
expect fmt 11 10 12 YU12 YV12 NV12 NV21 RGBP RGB3 BGR3 BGR4 YUYV YVYU UYVY VYUY
expect fmt 11 9 2 H264 MJPG
expect fmt 12 10 53 BA81 RGGB GBRG GRBG pBAA
expect fmt 12 9 13 YUYV YVYU VYUY UYVY YU12 YV12 RGB3 BGR3 AB24 BGR4 RGBP NV12 NV21
expect fmt 13 2 53 BA81 RGGB GBRG GRBG pBAA pRAA pGAA pgAA pBCC pRCC pGCC pgCC pBEE pREE pGEE pgEE BYR2 RG16 GB16 GR16 GREY Y10P Y12P Y14P Y16 YUYV YVYU VYUY UYVY YU12 YV12 RGB3 BGR3 XB24 XR24 RGBP NV12 NV21 BG10 RG10 GB10 BA10 BG12 RG12 GB12 BA12 BG14 RG14 GB14 GR14 Y10 Y12 Y14
expect fmt 14 1 13 YUYV YVYU VYUY UYVY YU12 YV12 RGB3 BGR3 XB24 XR24 RGBP NV12 NV21
expect fmt 15 1 8 YUYV YVYU VYUY UYVY YU12 YV12 NV12 NV21
expect fmt 16 13 1 BSTA
expect fmt 18 10 2 YU12 NC12
expect fmt 18 9 2 YU12 NC12
expect fmt 31 10 12 YU12 YV12 NV12 NV21 RGBP RGB3 BGR3 BGR4 YUYV YVYU UYVY VYUY
expect fmt 31 9 1 JPEG
expect pi@raspberrypi:~$

# The board powers off: RSTS asks the firmware not to boot, smirror
# ends. Linux waits for the mini UART for two bytes' time only, so the
# end of its last line is lost, as on the board.
type sudo poweroff\r
expect password for pi:
type raspberry\r
timeout 900
expect reboot: Power
expect the board halted
end
